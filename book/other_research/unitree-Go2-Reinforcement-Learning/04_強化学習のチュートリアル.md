# 4. 強化学習チュートリアル

## 概要

本章では構築した開発環境を用いてUnitree Go2の強化学習を始めるためのチュートリアルを実施する．本章を終えることで，Isaac Sim上でUnitree Go2の強化学習を実行することができる．

## 前提条件

2章の環境構築を実施済みであること．本章では，学習用PC（NVIDIA GPU搭載，Ubuntu 24.04）で作業する．

## 理論背景

### 強化学習とは

**強化学習**とは「エージェントが**環境**と**相互作用**しながら試行錯誤を繰り返し，**報酬**を最大化する**方策**を学ぶ機械学習手法の一つ」である．**教師あり学習**は正解ラベル付きのデータを学習させて未知のデータに対する正解を予測する．**教師なし学習**は正解ラベルの付いていないデータから内在するパターンなどをコンピュータが独自に学習していく．それらに対して**強化学習**はデータではなく**環境**が与えられる．その後，学習したい行動に対して設定された**報酬**がより多く得られるようにコンピュータが自ら学習していく．以上が，強化学習と教師あり学習・教師なし学習との違いである．

### MDP（マルコフ決定過程）

MDPは，マルコフ性という，次の状態と報酬は現在の状態と行動のみに依存すると仮定した，逐次的な意思決定問題をモデル化するものである．MDPは，状態，行動，報酬，遷移確率の4つの要素から構成される．

### PPO（Proximal Policy Optimization）

方策ベースの強化学習アルゴリズムで，Actor-Critic構造を採用したオンポリシー型強化学習手法．損失を計算する際に，方策が大きく変化しないように抑制するため，クリッピング処理を行う．これにより，学習の安定性が向上する．

```python
class BasePPORunnerCfg(RslRlOnPolicyRunnerCfg):
    num_steps_per_env = 24          # 1回の更新前に，各環境で収集するステップ数　学習速度を上げるか学習精度を上げるかのバランスを決める
    max_iterations = 10000          # 学習全体の最大イテレーション（更新）回数
    save_interval = 100             # 何イテレーションごとにモデルを保存するか　学習結果に直接関係なし
    experiment_name = ""            # 実験名（タスク名と同じにした方がよし）
    empirical_normalization = False # 観測値の経験的正規化（running mean/std）を使うかどうか　観測値が各モータ・センサ等でバラバラならTrueにすることを検討する

    policy = RslRlPpoActorCriticCfg(
        init_noise_std=1.0,                    # 方策の初期探索ノイズの標準偏差　(初期にどれだけランダムに行動させるか)
        actor_hidden_dims=[512, 256, 128],      # Actor（方策）ネットワークの隠れ層サイズ
        critic_hidden_dims=[512, 256, 128],     # Critic（価値関数）ネットワークの隠れ層サイズ
        activation="elu",                       # 各層で使う活性化関数
    )

    algorithm = RslRlPpoAlgorithmCfg(
        value_loss_coef=1.0,        # 価値関数損失に掛ける係数　大きいほど状態価値関数が方策に深くかかわる
        use_clipped_value_loss=True,# 価値損失にもPPOのクリッピングを適用するか
        clip_param=0.2,             # PPOの方策比クリッピング範囲　小さいほど学習は安定するが，遅くなる
        entropy_coef=0.01,          # エントロピーボーナスの係数（探索を促進）方策が選ぶ行動の確率分布がばらけているほど損失が少なくなるよう調整される．
        num_learning_epochs=5,      # 収集したデータに対して何エポック学習を回すか　(復習する回数)
        num_mini_batches=4,         # 1エポックあたりのミニバッチ分割数
        learning_rate=1.0e-3,       # 学習率の初期値
        schedule="adaptive",        # 学習率スケジュール（KLに応じて適応的に調整）
        gamma=0.99,                 # 割引率（将来報酬の重み）
        lam=0.95,                   # 実測報酬をどれくらい"先のステップ"まで信頼して積み上げ，どこから先をCriticの予測に切り替えるか，そのバランスを連続的に調整するもの
        desired_kl=0.01,            # adaptiveスケジュール使用時の目標KL　(方策更新前後のKLダイバージェンスがこの値を超えると学習率減少，下回ると学習率増加)
        max_grad_norm=1.0,          # 勾配の大きさがクリッピングされない最大値（これを超えると勾配の大きさが強制的に1.0に）
    )
   
```

#### tasks/locomotion/mdp/rewards.py

このスクリプトでは報酬関数の部品を定義する関数をまとめている．自分で新たに報酬，損失を定義したいのならここに記述．Isaac Labが提供する関数・クラスが多く使われているため，以下のドキュメントを参照するとよい．

[isaaclab.envs.mdp.rewards — Isaac Lab Documentation](https://isaac-sim.github.io/IsaacLab/main/_modules/isaaclab/envs/mdp/rewards.html)

```python
#すべて記述すると長すぎるため一部を記載
from __future__ import annotations

import torch
from typing import TYPE_CHECKING

try:
    from isaaclab.utils.math import quat_apply_inverse
except ImportError:
    from isaaclab.utils.math import quat_rotate_inverse as quat_apply_inverse
from isaaclab.assets import Articulation, RigidObject
from isaaclab.managers import SceneEntityCfg
from isaaclab.sensors import ContactSensor

if TYPE_CHECKING:
    from isaaclab.envs import ManagerBasedRLEnv

"""
関節（Joint）に関するペナルティ
"""

def energy(env: ManagerBasedRLEnv, asset_cfg: SceneEntityCfg = SceneEntityCfg("robot")) -> torch.Tensor:
    """ロボットの関節が消費するエネルギーへのペナルティ"""
    # 仕事率(パワー) = |角速度| × |トルク| の総和を計算し，無駄な力み・エネルギー消費を抑制する
    asset: Articulation = env.scene[asset_cfg.name]

    qvel = asset.data.joint_vel[:, asset_cfg.joint_ids]
    qfrc = asset.data.applied_torque[:, asset_cfg.joint_ids]
    return torch.sum(torch.abs(qvel) * torch.abs(qfrc), dim=-1)

def stand_still(
    env: ManagerBasedRLEnv, command_name: str = "base_velocity", asset_cfg: SceneEntityCfg = SceneEntityCfg("robot")
) -> torch.Tensor:
    """静止指令が出ているのに動いてしまっていることへのペナルティ"""
    asset: Articulation = env.scene[asset_cfg.name]

    # 関節角度が初期姿勢からどれだけズレているかを計算
    reward = torch.sum(torch.abs(asset.data.joint_pos - asset.data.default_joint_pos), dim=1)
    # 速度指令がほぼゼロ（＝止まれと言われている）ときだけ，このペナルティを有効にする
    cmd_norm = torch.norm(env.command_manager.get_command(command_name), dim=1)
    return reward * (cmd_norm < 0.1)

"""
胴体（Robot）の姿勢に関する報酬
"""

def orientation_l2(
    env: ManagerBasedRLEnv, desired_gravity: list[float], asset_cfg: SceneEntityCfg = SceneEntityCfg("robot")
) -> torch.Tensor:
    """Unitreeを見た見かけの重力方向から姿勢を推定し，理想の姿勢とのズレの少なさを評価する報酬（L2二乗カーネル）"""
    # extract the used quantities (to enable type-hinting)
    asset: RigidObject = env.scene[asset_cfg.name]

    desired_gravity = torch.tensor(desired_gravity, device=env.device)
    # 実際の重力方向ベクトルと，理想の重力方向のコサイン類似度（内積）を計算
    cos_dist = torch.sum(asset.data.projected_gravity_b * desired_gravity, dim=-1)  # cosine distance
    # コサイン類似度は[-1, 1]の範囲なので，扱いやすい[0, 1]に変換する
    normalized = 0.5 * cos_dist + 0.5  # map from [-1, 1] to [0, 1]
    return torch.square(normalized)

def upward(env: ManagerBasedRLEnv, asset_cfg: SceneEntityCfg = SceneEntityCfg("robot")) -> torch.Tensor:
    """胴体が直立しているほど大きくなる報酬"""
    # extract the used quantities (to enable type-hinting)
    asset: RigidObject = env.scene[asset_cfg.name]
    # z成分が−1（直立）に近いほど値が大きくなる
    reward = torch.square(1 - asset.data.projected_gravity_b[:, 2])
    return reward

def joint_position_penalty(
    env: ManagerBasedRLEnv, asset_cfg: SceneEntityCfg, stand_still_scale: float, velocity_threshold: float
) -> torch.Tensor:
    """関節角度が初期姿勢からズレていることへのペナルティ（動いているかどうかで強さを変える）"""
    # extract the used quantities (to enable type-hinting)
    asset: Articulation = env.scene[asset_cfg.name]
    cmd = torch.linalg.norm(env.command_manager.get_command("base_velocity"), dim=1)
    body_vel = torch.linalg.norm(asset.data.root_lin_vel_b[:, :2], dim=1)
    reward = torch.linalg.norm((asset.data.joint_pos - asset.data.default_joint_pos), dim=1)
    # 指令があるか，実際に動いていれば通常のペナルティ
    # 静止すべき状況（指令ゼロ かつ 実速度も小さい）なら，stand_still_scale倍に強化する
    return torch.where(torch.logical_or(cmd > 0.0, body_vel > velocity_threshold), reward, stand_still_scale * reward)
```

##### **COBBLESTONE_ROAD_CFG：**

```python
# ===== 地形生成の設定 =====
# 1枚の大きな学習用地形を「8m四方のタイル × 10行 × 20列」で生成する
#   行（row）→ 難易度（上の行ほど易しく，下の行ほど難しい）※カリキュラム有効時
#   列（col）→ 地形の種類（proportion の比率で割り振り，環境ごとに固定）
# 世界は1つだけ．並列環境のロボットは全員，この地形上のどれかのタイルに配置される
COBBLESTONE_ROAD_CFG = terrain_gen.TerrainGeneratorCfg(
    size=(8.0, 8.0),  # 1タイルの大きさ [m]（全体は 10行×8m = 80m，20列×8m = 160m）
    border_width=20.0,  # 地形全体を囲む平地の幅 [m]．
    num_rows=10,  # 行数 ＝ 難易度の段階数
    num_cols=20,  # 列数 ＝ 地形の種類を並べる数（proportion に従って割り当て）
    horizontal_scale=0.1,  # 高さマップの格子間隔 [m]．Hf系の地形にだけ効く（高さスキャナーの間隔と同じ）
    vertical_scale=0.005,  # 高さマップの高さの最小刻み [m]（5mm）．Hf系の地形にだけ効く
    slope_threshold=0.75,  # 高さマップ→メッシュ変換時，傾きがこれ（約37°）を超える部分を垂直な壁に直す
    difficulty_range=(0.0, 1.0),  # 難易度の範囲．
    use_cache=False,  # 生成した地形をディスクに保存して再利用するか（False: 毎回生成し直す）
    sub_terrains={
```

##### **ObservationsCfg：**

方策ニューラルネットワークへの入力と価値関数に用いる正確な観測を定義しているクラス．子クラスとして**PolicyCfg**と**CriticCfg**が定義されている．各子クラスで設定されている項目は以下の通り．

PolicyCfg：

- **方策**へ入力するセンサー・モーターが取得した情報
- 入力する情報のノイズ幅

CriticCfg：

- **状態価値関数**へ入力するセンサー・モーターが取得した正解(ノイズなし)情報
- 仮想的なセンサー(height_scan等もこっち)

```python
class ObservationsCfg:
    """Observation specifications for the MDP."""
    # ポリシー(実際にロボットを動かす方策)と，学習時だけに使うcritic(価値関数)とで
    # 観測する内容を分けて定義している

    @configclass
    class PolicyCfg(ObsGroup):
        """Observations for policy group."""
        # 実機に載せる方策(actor)への入力．ノイズを混ぜているのは，
        # 実際のセンサ誤差を模擬してsim2realのギャップを埋めるため

        # observation terms (order preserved)

        # 胴体の角速度(roll/pitch/yaw方向の回転速度)
        # scale=0.2 : ネットワークに渡す前に値を0.2倍してスケール調整(学習を安定させるための正規化)
        # clip=(-100,100) : 異常値の飽和防止(実質ほぼ効かない緩い制限)
        # noise=Unoise(±0.2) : 一様分布のノイズを±0.2の範囲で加える(センサ誤差の模擬)
        base_ang_vel = ObsTerm(func=mdp.base_ang_vel, scale=0.2, clip=(-100, 100), noise=Unoise(n_min=-0.2, n_max=0.2))

        # 重力方向を胴体のローカル座標系に投影したベクトル
        # 胴体がどれだけ傾いているか(絶対姿勢そのものではなく重力に対する向き)を表す
        # IMU由来の情報を模擬しているため，ノイズは小さめ(±0.05)
        projected_gravity = ObsTerm(func=mdp.projected_gravity, clip=(-100, 100), noise=Unoise(n_min=-0.05, n_max=0.05))

        # 現在の速度指令(コマンド)．CommandsCfgで定義された"base_velocity"の値をそのまま観測に含める
        # これは内部で生成している指令値そのものなのでノイズは付けていない(正確な値として与える)
        velocity_commands = ObsTerm(
            func=mdp.generated_commands, clip=(-100, 100), params={"command_name": "base_velocity"}
        )

        # 各関節角度の，初期姿勢(デフォルト角度)からの相対値
        # 関節エンコーダの読み取り誤差を模擬した小さいノイズ(±0.01)を付加
        joint_pos_rel = ObsTerm(func=mdp.joint_pos_rel, clip=(-100, 100), noise=Unoise(n_min=-0.01, n_max=0.01))

        # 各関節角速度の相対値．scale=0.05で値を縮小
        # 角速度は生の値が大きくなりがちなノイズ源のため，ノイズ範囲も広め(±1.5)に設定
        joint_vel_rel = ObsTerm(
            func=mdp.joint_vel_rel, scale=0.05, clip=(-100, 100), noise=Unoise(n_min=-1.5, n_max=1.5)
        )

        # 直前ステップで方策が出力した行動(目標関節角度指令)
        # 過去の自分の行動を観測に含めることで，行動の連続性・滑らかさを学習しやすくする
        last_action = ObsTerm(func=mdp.last_action, clip=(-100, 100))

        def __post_init__(self):
            # self.history_length = 5
            # ↑コメントアウト中：過去5ステップ分を履歴として積む(フレームスタック)設定．
            #   現状は無効化されており，各ステップ単発の観測のみを使っている

            # ノイズ付加(上で設定したUnoiseなど)を実際に有効化するかどうかのスイッチ
            self.enable_corruption = True

            # 上記の各ObsTermをバラバラなテンソルのままにせず，1本のベクトルに連結するかどうか
            # True にすることで，ニューラルネットへの入力としてそのまま渡せる1次元ベクトルになる
            self.concatenate_terms = True

    # observation groups
    # PolicyCfgのインスタンスを"policy"という名前でこのCfg全体に登録
    policy: PolicyCfg = PolicyCfg()

    @configclass
    class CriticCfg(ObsGroup):
        """Observations for critic group."""
        # 価値関数(critic)専用の観測グループ．学習時のみ使用し，実機の方策には使われない
        # (asymmetric actor-critic：criticだけがシミュレーション上の真の値を観測できる)

        # 胴体の並進速度(x,y,z)．実機では正確に測るのが難しい(privileged)情報だが，
        # シミュレーション上は正解値をそのまま渡せるためcriticにだけ含めている
        base_lin_vel = ObsTerm(func=mdp.base_lin_vel, clip=(-100, 100))

        # 以下，policy側とほぼ同じ項目群だが，ノイズ(Unoise)が一切付いていない点が大きな違い
        # → criticは「本当の値」をそのまま見て正確に価値を推定するのが役目なので，
        #   センサ誤差を模擬する必要がない
        base_ang_vel = ObsTerm(func=mdp.base_ang_vel, scale=0.2, clip=(-100, 100))
        projected_gravity = ObsTerm(func=mdp.projected_gravity, clip=(-100, 100))
        velocity_commands = ObsTerm(
            func=mdp.generated_commands, clip=(-100, 100), params={"command_name": "base_velocity"}
        )
        joint_pos_rel = ObsTerm(func=mdp.joint_pos_rel, clip=(-100, 100))
        joint_vel_rel = ObsTerm(func=mdp.joint_vel_rel, scale=0.05, clip=(-100, 100))

        # 各関節が実際に発生させているトルク(努力量)．これも実機では推定が難しいprivileged情報
        joint_effort = ObsTerm(func=mdp.joint_effort, scale=0.01, clip=(-100, 100))

        last_action = ObsTerm(func=mdp.last_action, clip=(-100, 100))

       
        #height_scanner = ObsTerm(func=mdp.height_scan,
        #    params={"sensor_cfg": SceneEntityCfg("height_scanner")},
        #    clip=(-1.0, 5.0),
        #)

        # def __post_init__(self):
        #     self.history_length = 5
        # ↑policy側と同様，履歴フレームスタックの設定だが，
        #   こちらは__post_init__自体が丸ごとコメントアウトされているため，
        #   enable_corruption/concatenate_termsのデフォルト値がそのまま使われる

    # privileged observations
    # CriticCfgのインスタンスを"critic"という名前で登録．rsl_rl側が
    # actor(policy)とcritic(critic)それぞれに異なる観測を渡すために使う
    critic: CriticCfg = CriticCfg()
```
